const el = (tag, cls, text) => {
  const n = document.createElement(tag);
  if (cls) n.className = cls;
  if (text !== undefined) n.textContent = text;

async function api(path, opts) {
  const res = await fetch(path, opts);
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw Object.assign(new Error(data.error || res.statusText), { status: res.status, data });
  return data;
}

let currentEngine = null;

async function loadEngines() {
  const { engines, default: def } = await api("/api/engines");
  const sel = $("#engineSelect");
  sel.innerHTML = "";
  engines.forEach((e) => {
    const opt = el("option");
    opt.value = e.id;
    opt.textContent = e.configured ? e.label : `${e.label} (no key)`;
    opt.disabled = !e.configured;
    sel.appendChild(opt);
  });
  currentEngine = def;
  sel.value = def;
  sel.addEventListener("change", () => (currentEngine = sel.value));
}

function addMessage(role, text, engineTag) {
  const wrap = el("div", `msg ${role}`, text);
  if (engineTag) wrap.appendChild(el("span", "engine-tag", engineTag));
  if (role === "assistant") {
    const btn = el("button", "speak-btn", "🔊 Speak");
    btn.addEventListener("click", () => speak(text, btn));
    wrap.appendChild(document.createElement("br"));
    wrap.appendChild(btn);
  }
  $("#messages").appendChild(wrap);
  $("#messages").scrollTop = $("#messages").scrollHeight;
  return wrap;
}

async function speak(text, btn) {
  btn.classList.add("loading");
  btn.textContent = "…";
  try {
    const res = await fetch("/api/speak", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ text }),
    });
    if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error || "speak failed");
    const blob = await res.blob();
    const player = $("#player");
    player.src = URL.createObjectURL(blob);
    player.hidden = false;
    player.play();
  } catch (err) {
    alert("Speak failed: " + err.message);
  } finally {
    btn.classList.remove("loading");
    btn.textContent = "🔊 Speak";
  }
}

async function runShell(command, force, out) {
  try {
    const result = await api("/api/device/shell", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ command, force }),
    });
    out.textContent = `exit ${result.exit_code}\n${result.stdout || ""}${result.stderr ? "\nstderr:\n" + result.stderr : ""}`;
  } catch (err) {
    if (err.status === 409) {
      if (confirm(`"${command}" looks destructive. Run it anyway?`)) {
        return runShell(command, true, out);
      }
      out.textContent = "Cancelled.";
    } else {
      out.textContent = "Error: " + err.message;
    }
  }
}

=== device/termux_api.py ===
"""Wraps the `termux-api` command-line tools (from the Termux:API app +
`pkg install termux-api`) so Engola can actually touch the phone: battery,
clipboard, notifications, vibration, torch, volume, location, SMS, camera.

Each function shells out to one termux-api binary and returns plain text.
If the Termux:API app/package isn't installed, every call fails with a
clear message rather than hanging.
"""
import json
import shlex
import subprocess

TIMEOUT = 20

ACTIONS = {
    "battery": "termux-battery-status",
    "clipboard_get": "termux-clipboard-get",
    "torch_on": "termux-torch on",
    "torch_off": "termux-torch off",
    "vibrate": "termux-vibrate -d 300",
    "wifi": "termux-wifi-connectioninfo",
    "location": "termux-location -p gps -r once",
    "sms_list": "termux-sms-list -l 10",
    "volume": "termux-volume",
}


def _run(cmd: str) -> str:
    try:
        proc = subprocess.run(shlex.split(cmd), capture_output=True, text=True, timeout=TIMEOUT)
    except FileNotFoundError:
        return (
            "termux-api is not installed. On the phone: install the 'Termux:API' app "
            "from F-Droid/Play Store, then run `pkg install termux-api`."
        )
    except subprocess.TimeoutExpired:
        return f"{cmd} timed out after {TIMEOUT}s (check the Termux:API app has the needed permission)."
    out = (proc.stdout or proc.stderr or "").strip()
    return out or "(no output)"


def run_action(name: str) -> str:
    if name not in ACTIONS:
        return f"Unknown device action '{name}'. Options: {', '.join(sorted(ACTIONS))}, clipboard_set, notify, share."
    return _run(ACTIONS[name])


def clipboard_set(text: str) -> str:
    try:
        subprocess.run(["termux-clipboard-set"], input=text, text=True, timeout=TIMEOUT)
        return "Copied to clipboard."
    except FileNotFoundError:
        return "termux-api is not installed (see /device help)."


def notify(title: str, content: str) -> str:
    try:
        subprocess.run(
            ["termux-notification", "--title", title, "--content", content],
            timeout=TIMEOUT,
        )
        return "Notification sent."
    except FileNotFoundError:
        return "termux-api is not installed (see /device help)."


def sms_send(number: str, text: str) -> str:
    try:
        subprocess.run(["termux-sms-send", "-n", number, text], timeout=TIMEOUT)
        return f"SMS sent to {number}."
    except FileNotFoundError:
        return "termux-api is not installed (see /device help)."


def camera_photo(path: str) -> str:
    try:
        subprocess.run(["termux-camera-photo", "-c", "0", path], timeout=TIMEOUT)
        return path
    except FileNotFoundError:
        return "termux-api is not installed (see /device help)."


HELP = (
    "Device actions: " + ", ".join(sorted(ACTIONS)) + ", clipboard_set <text>, "
    "notify <title> | <text>, sms_send <number> <text>, photo\n"
    "Requires the Termux:API app installed + `pkg install termux-api`."
)


=== device/screen.py ===
"""'Read the screen' is not something a plain Termux shell can do on its
own -- Android sandboxes apps from each other, and there's no termux-api
command for it. The real, legitimate, no-root path is the same one
Android's own developer tools use: pair Termux to the device's own ADB
(Android Debug Bridge) over Wireless Debugging, which Google explicitly
built for exactly this kind of on-device automation and testing.

One-time setup on the phone (not automatable -- it's a deliberate consent
step by design):
  1. `pkg install android-tools` in Termux
  2. Settings -> Developer options -> Wireless debugging -> ON
  3. "Pair device with pairing code" -> note the IP:port and 6-digit code
  4. `adb pair <ip>:<port>` in Termux, enter the code
  5. `adb connect <ip>:<port>` (the regular debugging port, shown on the
     same screen) -- after this, `adb devices` should list it as authorized

This grants real control (screenshots AND simulated taps/typing), which is
strictly more powerful than the other device/ modules -- treat pairing it
as a deliberate, bigger trust decision than installing termux-api.
"""
import shlex
import subprocess

import config

TIMEOUT = 20
SCREENSHOT_PATH = str(config.DATA_DIR / "screen.png")


def _adb(*args) -> subprocess.CompletedProcess:
    return subprocess.run(["adb", *args], capture_output=True, timeout=TIMEOUT)


def connected() -> bool:
    try:
        out = _adb("devices").stdout.decode()
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return False
    lines = [l for l in out.splitlines()[1:] if l.strip()]
    return any(l.endswith("device") for l in lines)  # excludes 'unauthorized'/'offline'


def _require_connection():
    if not connected():
        raise RuntimeError(
            "No authorized adb device. Run `adb devices` in Termux -- if it's empty or says "
            "'unauthorized', redo the Wireless Debugging pairing (see device/screen.py docstring)."
        )


def screenshot() -> bytes:
    _require_connection()
    proc = subprocess.run(["adb", "exec-out", "screencap", "-p"], capture_output=True, timeout=TIMEOUT)
    if not proc.stdout:
        raise RuntimeError(f"Screenshot failed: {proc.stderr.decode()[:300]}")
    with open(SCREENSHOT_PATH, "wb") as fh:
        fh.write(proc.stdout)
    return proc.stdout


def read_screen() -> str:
    """Screenshot + Gemini vision -> a text description of what's on screen,
    including any readable text. This is the actual 'reads your screen'
    feature -- OCR-via-vision-model, not a live accessibility feed."""
    image = screenshot()
    from providers import gemini
    return gemini.describe_image(image)


def tap(x: int, y: int) -> str:
    _require_connection()
    _adb("shell", "input", "tap", str(x), str(y))
    return f"Tapped ({x}, {y})"


def swipe(x1: int, y1: int, x2: int, y2: int, duration_ms: int = 300) -> str:
    _require_connection()
    _adb("shell", "input", "swipe", str(x1), str(y1), str(x2), str(y2), str(duration_ms))
    return f"Swiped ({x1},{y1}) -> ({x2},{y2})"


def type_text(text: str) -> str:
    _require_connection()
    # adb input text needs spaces escaped
    _adb("shell", "input", "text", shlex.quote(text).replace(" ", "%s"))
    return "Typed."


def tap_key(keyevent: str) -> str:
    """E.g. 'KEYCODE_BACK', 'KEYCODE_HOME', 'KEYCODE_ENTER'."""
    _require_connection()
    _adb("shell", "input", "keyevent", keyevent)
    return f"Sent {keyevent}"


=== device/shell.py ===
"""Runs shell commands on the device Engola itself is running on.

This is deliberately simple: the owner-ID check in bot.py is the real
access control (only Telegram messages from OWNER_TELEGRAM_ID ever reach
this module at all). The pattern check below is a second, much smaller
safety net against fat-fingering an obviously destructive one-liner --
not a sandbox, and not a substitute for keeping the bot token private.
"""
import re
import subprocess

import config

MAX_OUTPUT = 3500


def is_dangerous(command: str) -> bool:
    return any(re.search(p, command, re.IGNORECASE) for p in config.DANGEROUS_PATTERNS)


def run(command: str, timeout: int | None = None) -> dict:
    timeout = min(timeout or config.MAX_SHELL_TIMEOUT, 600)
    try:
        proc = subprocess.run(
            command, shell=True, capture_output=True, text=True, timeout=timeout,
        )
        return {
            "exit_code": proc.returncode,
            "stdout": proc.stdout[-MAX_OUTPUT:],
            "stderr": proc.stderr[-MAX_OUTPUT:],
        }
    except subprocess.TimeoutExpired:
        return {"exit_code": -1, "stdout": "", "stderr": f"Timed out after {timeout}s"}


def format_result(result: dict) -> str:
    parts = [f"exit code: {result['exit_code']}"]
    if result["stdout"]:
        parts.append(f"```\n{result['stdout']}\n```")
    if result["stderr"]:
        parts.append(f"stderr:\n```\n{result['stderr']}\n```")
    return "\n".join(parts)
