=== providers/registry.py ===
"""Registry of chat engines. Adding a fifth provider later means writing
providers/whatever.py with configured()/respond()/wants_search(), then
adding one line to ENGINES below -- nothing else in the app changes."""
import config
from providers import anthropic_api, gemini, grok, openai_api

ENGINES = {
    "gemini": {"label": "Gemini 2.0 Flash", "module": gemini},
    "grok": {"label": "Grok (x.ai)", "module": grok},
    "openai": {"label": "GPT (OpenAI)", "module": openai_api},
    "claude": {"label": "Claude (Anthropic)", "module": anthropic_api},
}


def available() -> list[dict]:
    return [
        {"id": key, "label": v["label"], "configured": v["module"].configured()}
        for key, v in ENGINES.items()
    ]


def default_engine() -> str:
    if config.DEFAULT_ENGINE in ENGINES and ENGINES[config.DEFAULT_ENGINE]["module"].configured():
        return config.DEFAULT_ENGINE
    for key, v in ENGINES.items():
        if v["module"].configured():
            return key
    return config.DEFAULT_ENGINE  # none configured -- caller will surface the resulting error


def respond(engine_id: str, history: list[dict], user_text: str, extra_context: str = "", allow_search_request: bool = False) -> str:
    if engine_id not in ENGINES:
        raise RuntimeError(f"Unknown engine '{engine_id}'. Options: {', '.join(ENGINES)}")
    return ENGINES[engine_id]["module"].respond(history, user_text, extra_context, allow_search_request)


def wants_search(engine_id: str, reply: str) -> str | None:
    return ENGINES[engine_id]["module"].wants_search(reply)


=== providers/openai_compat.py ===
"""Shared client for any OpenAI-compatible /chat/completions API.
Grok (x.ai) and OpenAI itself both speak this exact format, so one
function serves both providers instead of duplicating the HTTP plumbing."""
import requests

SEARCH_TAG = "SEARCH:"


def _messages_from_history(system: str, history: list[dict], user_text: str, extra_context: str = ""):
    messages = [{"role": "system", "content": system}]
    for turn in history:
        role = "assistant" if turn["role"] == "assistant" else "user"
        messages.append({"role": role, "content": turn["content"]})
    text = user_text if not extra_context else f"{extra_context}\n\n---\n\n{user_text}"
    messages.append({"role": "user", "content": text})
    return messages


def respond(base_url: str, api_key: str, model: str, system: str, history: list[dict],
            user_text: str, extra_context: str = "", allow_search_request: bool = False) -> str:
    if not api_key:
        raise RuntimeError("This provider has no API key configured")

    if allow_search_request:
        system = system + (
            f"\nIf, and only if, answering well truly requires current/live information you "
            f"don't have, reply with EXACTLY one line: '{SEARCH_TAG} <best web search query>' "
            f"and nothing else. Otherwise just answer normally."
        )

    payload = {
        "model": model,
        "messages": _messages_from_history(system, history, user_text, extra_context),
        "temperature": 0.6,
        "max_tokens": 1500,
    }
    r = requests.post(
        f"{base_url}/chat/completions",
        json=payload,
        headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
        timeout=45,
    )
    if r.status_code == 404:
        raise RuntimeError(f"Model '{model}' not found at {base_url} -- check the provider's current model list.")
    r.raise_for_status()
    data = r.json()
    choices = data.get("choices") or []
    if not choices:
        raise RuntimeError(f"No response from {base_url}: {data}")
    content = (choices[0].get("message") or {}).get("content", "").strip()
    if not content:
        raise RuntimeError("Provider returned an empty answer")
    return content


def wants_search(reply: str) -> str | None:
    stripped = reply.strip()
    if stripped.startswith(SEARCH_TAG):
        return stripped[len(SEARCH_TAG):].strip().strip('"')
    return None


=== providers/telegram.py ===
"""Send-only Telegram client. In v1 Telegram was the whole UI; in v2 the
web page is the UI and this is purely an archive/notification channel --
so no long-polling, no getUpdates, just sendMessage/sendDocument."""
import io

import requests

import config

API = f"https://api.telegram.org/bot{config.TELEGRAM_BOT_TOKEN}"


def configured() -> bool:
    return bool(config.TELEGRAM_BOT_TOKEN and config.ARCHIVE_CHAT_ID)


def send_message(text: str) -> None:
    if not configured():
        return
    for i in range(0, len(text), 4000):
        try:
            requests.post(f"{API}/sendMessage", json={
                "chat_id": config.ARCHIVE_CHAT_ID, "text": text[i:i + 4000], "parse_mode": "Markdown",
            }, timeout=15)
        except Exception:
            pass


def send_document_path(path: str, caption: str = "") -> None:
    if not configured():
        return
    try:
        with open(path, "rb") as fh:
            requests.post(
                f"{API}/sendDocument",
                data={"chat_id": config.ARCHIVE_CHAT_ID, "caption": caption[:1024]},
                files={"document": (path.split("/")[-1], fh)},
                timeout=30,
            )
    except Exception:
        pass


=== memory/archive.py ===
"""Every exchange gets written to a local daily markdown transcript. If
Telegram archiving is configured (TELEGRAM_BOT_TOKEN + ARCHIVE_CHAT_ID),
it's also pushed there live plus as a full .md file backup periodically --
same behavior as v1, just no longer doubling as the chat UI itself."""
from datetime import datetime

import config
from providers import telegram

ARCHIVE_FILE_EVERY = 20
_counter = {"n": 0}


def _today_path():
    day = datetime.now().strftime("%Y-%m-%d")
    return config.LOG_DIR / f"{day}.md", day


def log_turn(user_text: str, assistant_text: str, engine: str = "") -> None:
    path, day = _today_path()
    ts = datetime.now().strftime("%H:%M:%S")
    tag = f" _{engine}_" if engine else ""
    entry = f"\n### {ts}{tag}\n\n**{config.OWNER_NAME}:** {user_text}\n\n**Engola:** {assistant_text}\n"
    is_new = not path.exists()
    with open(path, "a", encoding="utf-8") as fh:
        if is_new:
            fh.write(f"# Engola transcript -- {day}\n")
        fh.write(entry)

    telegram.send_message(f"🗒 *{ts}*\n*{config.OWNER_NAME}:* {user_text}\n*Engola:* {assistant_text}")

    _counter["n"] += 1
    if _counter["n"] % ARCHIVE_FILE_EVERY == 0:
        push_today_file()


def push_today_file() -> None:
    path, day = _today_path()
    if path.exists():
        telegram.send_document_path(str(path), caption=f"Transcript backup -- {day}")


def log_event(text: str) -> None:
    path, day = _today_path()
    ts = datetime.now().strftime("%H:%M:%S")
    is_new = not path.exists()
    with open(path, "a", encoding="utf-8") as fh:
        if is_new:
            fh.write(f"# Engola transcript -- {day}\n")
        fh.write(f"\n> **{ts} [event]** {text}\n")
